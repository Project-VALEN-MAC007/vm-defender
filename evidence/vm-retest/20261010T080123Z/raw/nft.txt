table inet adaptive_defender {
	set ssh_redirect {
		type ipv4_addr
		timeout 30m
	}

	set telnet_redirect {
		type ipv4_addr
		timeout 30m
	}

	chain prerouting {
		type nat hook prerouting priority dstnat; policy accept;
		iifname "trap-out" ip saddr @ssh_redirect tcp dport 22 counter packets 5 bytes 300 dnat ip to 10.10.10.2:2222
		iifname "trap-out" ip saddr @telnet_redirect tcp dport 23 counter packets 2 bytes 120 dnat ip to 10.10.10.2:2223
	}

	chain forward {
		type filter hook forward priority filter; policy drop;
		ct state invalid counter packets 0 bytes 0 drop
		ct state established,related counter packets 169 bytes 31117 accept
		iifname "trap-out" oifname "enp0s8" ip daddr 10.10.10.2 tcp dport 2222 ct status dnat counter packets 5 bytes 300 accept
		iifname "trap-out" oifname "enp0s8" ip daddr 10.10.10.2 tcp dport 2223 ct status dnat counter packets 2 bytes 120 accept
	}
}
